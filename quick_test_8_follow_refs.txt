#[test]
fn test_follow_west_leads_S6_east_follows_2nd_nt() {
    let hands = get_quick_test_8_hands();
    let lead_suit = S;
    let playable = hands[EAST].suit(lead_suit);
    let playable = if playable.is_empty() { hands[EAST] } else { playable };
    let ordered = order_follows(
        playable,
        &hands,
        EAST,
        NOTRUMP,
        S,  // lead_suit
        WEST,  // winning_seat
        card_of(S, 4),  // winning_card: S6
        1,  // card_in_trick
        |c1, c2| wins_over(c1, c2, NOTRUMP),
    );
    let result = ordered_to_string(ordered.iter());
    assert_eq!(result, "SJ S7");
}
